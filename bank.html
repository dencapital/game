Банк → игра

Настраивается один раз, минут пять. Дальше iPhone сам отправляет каждую банковскую SMS в твою финансовую таблицу: сумма, время, куда ушли деньги — определяются автоматически. Тебе остаётся заполнить одну колонку — «на что».

iOS не даёт ни одному приложению читать SMS — это ограничение самой системы, а не игры. Поэтому цепочка: SMS → «Команды» на iPhone → твоя таблица → робот дважды в день → игра.

1Скрипт в таблице

  1. Открой свою таблицу личных финансов на компьютере.
  2. Меню Расширения → Apps Script.
  3. Сотри всё, что там есть, и вставь код (кнопка ниже копирует его целиком).
  5. Сохрани (⌘S). Затем Начать развёртывание → Новое развёртывание.
  6. Тип — Веб-приложение. «Запуск от имени» — я. «У кого есть доступ» — у всех.
  7. «Развернуть» → Google спросит разрешение: Дополнительные настройки → Перейти к проекту → Разрешить.
  8. Скопируй URL веб-приложения — длинный, оканчивается на /exec. Он понадобится на шаге 2.

«Доступ у всех» = по этому адресу можно только дописать строку. Прочитать таблицу или что-то удалить через него нельзя. Просто не публикуй адрес.

Показать код 
/** Банковские SMS → лист «SMS_БАНК» этой таблицы. Ставится через Расширения → Apps Script. */
var TAB  = 'SMS_БАНК';
var HEAD = ['время','банк','сумма','валюта','тип','контрагент','НА ЧТО (заполняешь ты)','остаток после','текст SMS'];

function doPost(e) {
  var p = (e && e.parameter) || {};
  var text = p.text || '', sender = p.sender || '';
  if (!text) return ContentService.createTextOutput('empty');

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(TAB);
  if (!sh) {
    sh = ss.insertSheet(TAB);
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEAD.length).setFontWeight('bold');
    sh.setColumnWidth(7, 220);
    sh.setColumnWidth(9, 420);
  }

  // дедуп: та же SMS за последние 5 минут (банки любят дублировать)
  var last = sh.getLastRow();
  if (last > 1) {
    var n = Math.min(20, last - 1);
    var prev = sh.getRange(last - n + 1, 1, n, HEAD.length).getValues();
    var now = new Date().getTime();
    for (var i = 0; i < prev.length; i++) {
      var t0 = (prev[i][0] instanceof Date) ? prev[i][0].getTime() : 0;
      if (prev[i][8] === text && now - t0 < 5 * 60 * 1000) return ContentService.createTextOutput('dup');
    }
  }

  var r = parseSms(text);
  sh.appendRow([new Date(), sender, r.amount, r.cur, r.type, r.who, '', r.bal, text]);
  return ContentService.createTextOutput('ok');
}

function parseSms(s) {
  var t = String(s).replace(/ /g, ' ');
  var cur = /IDR|\bRp\b/i.test(t) ? 'IDR'
          : /USD|\$/.test(t)      ? 'USD'
          : /RUB|руб|₽/i.test(t)  ? 'RUB'
          : /THB|บาท/i.test(t)    ? 'THB' : '';
  var type = /(kredit|masuk|diterima|incoming|refund|зачислен|поступлен|пополнен|возврат)/i.test(t) ? 'приход'
           : /(debit|tarik|pembelian|pembayaran|transfer|purchase|payment|withdraw|оплата|покупка|списан|снятие|перевод)/i.test(t) ? 'расход' : '';
  return { amount: pickAmount(t), cur: cur, type: type, who: pickWho(t), bal: pickBalance(t) };
}

function toNum(x) {                      // 1.500.000,50 | 1,500,000.50 | 1500000
  x = String(x).trim().replace(/\s/g, '');
  if (/,\d{1,2}$/.test(x))      x = x.replace(/\./g, '').replace(',', '.');
  else if (/\.\d{1,2}$/.test(x)) x = x.replace(/,/g, '');
  else                           x = x.replace(/[.,]/g, '');
  var v = parseFloat(x.replace(/[^\d.]/g, ''));
  return isNaN(v) ? '' : v;
}

function pickAmount(t) {
  var re = /(?:IDR|Rp|USD|\$|RUB|₽|руб\.?|THB)\s*([\d][\d.,\s]{1,20})/ig, m, all = [];
  while ((m = re.exec(t))) all.push(toNum(m[1]));
  if (!all.length) {
    var re2 = /([\d][\d.,\s]{1,20})\s*(?:IDR|Rp|USD|RUB|₽|руб\.?|THB)/ig;
    while ((m = re2.exec(t))) all.push(toNum(m[1]));
  }
  if (all.length) return all[0];         // первая сумма в тексте = сама операция, дальше обычно остаток
  var m3 = t.match(/[\d][\d.,]{3,20}/);
  return m3 ? toNum(m3[0]) : '';
}

function pickBalance(t) {
  var m = t.match(/(?:saldo|sisa|balance|остаток|доступно)[^\d]{0,25}([\d][\d.,\s]{1,20})/i);
  return m ? toNum(m[1]) : '';
}

function pickWho(t) {
  var m = t.match(/\b(?:di|at|to|ke|kepada|в|на)\s+([A-ZА-Я0-9][^\n,.;]{2,30})/);
  if (m) return m[1].trim();
  var m2 = t.match(/\b([A-Z][A-Z0-9&'\-\. ]{3,25})\b/);
  return m2 ? m2[1].trim() : '';
}

/** Проверка без телефона: запусти эту функцию — в лист упадёт тестовая строка. */
function ТЕСТ() {
  doPost({ parameter: {
    sender: 'ТЕСТ',
    text: 'Pembelian dgn Kartu Debit di ALFAMART BALI IDR 250.000 pada 14:32. Saldo IDR 12.480.000'
  }});
}

2Автоматизация на iPhone

  1. Приложение «Команды» → вкладка «Автоматизация» → +.
  2. Триггер — «Сообщение».
  3. Отправитель — имя банка, как он подписан в SMS (например Permata, Mandiri, BCA, Sber). На каждый банк — своя автоматизация, они складываются в одну таблицу.
  4. Внизу выбери «Запускать сразу», тумблер «Спрашивать перед запуском» — выключить.
  5. Далее → Новая пустая команда → Добавить действие → «Получить содержимое URL».
  6. В действии: URL — тот самый адрес из шага 1. Разверни стрелку «Показать больше»:
    • Метод — POST
    • Тело запроса — Форма
    • поле text → значение — переменная «Сообщение»
    • поле sender → значение — переменная «Отправитель»
  7. Готово.

Проверить без ожидания SMS: в Apps Script выбери функцию ТЕСТ и нажми «Выполнить» — в таблице появится лист SMS_БАНК с разобранной тестовой строкой.

3Что дальше делает робот

Дважды в день — утром в 08:00 и вечером в 21:00 — он читает лист SMS_БАНК и:

  • складывает суммы в доход/расход текущего месяца — блок «Месяц ±» на экране «Сегодня» становится живым;
  • переводит IDR в доллары по курсу 18 тыс;
  • ловит нештатное — крупное списание, превышение лимита $2 800/мес, непонятную операцию — и выносит это в «три действия на сегодня»;
  • то, что ты подписал в колонке «НА ЧТО», использует как категорию — со временем расходы начнут раскладываться сами.

Колонку «НА ЧТО» можно не заполнять каждый день — подписывай пачкой раз в неделю, робот подхватит задним числом.
